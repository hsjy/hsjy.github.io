抓取物体的代码示例：
```python
import numpy as np
import asyncio
import omni.kit.app

from omni.isaac.core import World
from omni.isaac.franka import Franka
from omni.isaac.core.objects import DynamicCuboid
from omni.isaac.franka.controllers import PickPlaceController


async def run_pick_place():

    if World.instance() is None:
        world = World()
    else:
        World.instance().clear_instance()
        world = World()

    world.scene.add_default_ground_plane()

    robot = world.scene.add(
        Franka(
            prim_path="/World/Franka",
            name="franka"
        )
    )

    cube = world.scene.add(
        DynamicCuboid(
            prim_path="/World/Cube",
            position=np.array([0.5, 0.0, 0.05]),
            scale=np.array([0.04, 0.04, 0.04]),
            color=np.array([1, 0, 0])
        )
    )

    controller = PickPlaceController(
        name="pick_place_controller",
        gripper=robot.gripper,
        robot_articulation=robot
    )

    # 初始化仿真
    await world.initialize_simulation_context_async()

    # reset
    await world.reset_async()

    picking_position = cube.get_world_pose()[0] + np.array([0, 0, -0.02])
    placing_position = np.array([0.3, -0.3, 0.05])

    for i in range(1000):
        actions = controller.forward(
            picking_position=picking_position,
            placing_position=placing_position,
            current_joint_positions=robot.get_joint_positions()
        )

        robot.apply_action(actions)

        # 关键：执行 physics step
        world.step(render=True)

        await omni.kit.app.get_app().next_update_async()

    print("Pick and Place 完成")


asyncio.ensure_future(run_pick_place())
```
